const cosmicAudio = { ctx: null, master: null, musicTimer: null, musicOn: false, volume: 0.60 }; function initCosmicAudio() { if (!cosmicAudio.ctx) { const AudioCtx = window.AudioContext || window.webkitAudioContext; if (!AudioCtx) return false; cosmicAudio.ctx = new AudioCtx(); cosmicAudio.master = cosmicAudio.ctx.createGain(); cosmicAudio.master.gain.value = cosmicAudio.volume; cosmicAudio.master.connect(cosmicAudio.ctx.destination); } if (cosmicAudio.ctx.state === "suspended") { cosmicAudio.ctx.resume(); } return true; } function cosmicTone(freq, duration, type = "sine", volume = 0.12, endFreq = null) { if (!initCosmicAudio()) return; const ctx = cosmicAudio.ctx; const now = ctx.currentTime; const osc = ctx.createOscillator(); const gain = ctx.createGain(); osc.type = type; osc.frequency.setValueAtTime(freq, now); if (endFreq !== null) { osc.frequency.exponentialRampToValueAtTime( Math.max(1, endFreq), now + duration ); } gain.gain.setValueAtTime(0.001, now); gain.gain.exponentialRampToValueAtTime( Math.max(0.001, volume), now + 0.03 ); gain.gain.exponentialRampToValueAtTime( 0.001, now + duration ); osc.connect(gain); gain.connect(cosmicAudio.master); osc.start(now); osc.stop(now + duration + 0.03); } function startCosmicMusic() { if (!initCosmicAudio() || cosmicAudio.musicOn) return; cosmicAudio.musicOn = true; const melody = [220, 261.63, 329.63, 293.66, 196, 246.94, 293.66, 261.63]; let step = 0; cosmicAudio.musicTimer = setInterval(() => { if (!cosmicAudio.musicOn) return; const note = melody[step % melody.length]; cosmicTone(note, 1.1, "sine", 0.045); cosmicTone(note / 2, 1.5, "triangle", 0.025); step++; }, 850); } function stopCosmicMusic() { cosmicAudio.musicOn = false; if (cosmicAudio.musicTimer !== null) { clearInterval(cosmicAudio.musicTimer); cosmicAudio.musicTimer = null; } } function soundEngine() { cosmicTone(75, 0.35, "sawtooth", 0.07, 95); } function soundCollision() { cosmicTone(140, 0.4, "triangle", 0.25, 38); cosmicTone(65, 0.5, "sine", 0.18, 28); } function soundLevelUp() { cosmicTone(440, 0.18, "sine", 0.15, 660); setTimeout(() => { cosmicTone(660, 0.22, "sine", 0.15, 880); }, 150); setTimeout(() => { cosmicTone(880, 0.3, "sine", 0.13, 1100); }, 300); } function soundGameOver() { stopCosmicMusic(); cosmicTone(220, 0.6, "triangle", 0.22, 110); setTimeout(() => { cosmicTone(110, 1.0, "sine", 0.18, 45); }, 300); } function setCosmicVolume(value) { cosmicAudio.volume = Math.max(0, Math.min(1, value)); if (cosmicAudio.master && cosmicAudio.ctx) { cosmicAudio.master.gain.setTargetAtTime( cosmicAudio.volume, cosmicAudio.ctx.currentTime, 0.05 ); } }
