/* ===== COSMIC AUDIO SYSTEM ===== */ const cosmicAudio = { context: null, master: null, musicGain: null, engineGain: null, musicTimer: null, engineOsc: null, enabled: true, volume: 0.60 }; function initCosmicAudio() { const AudioCtx = window.AudioContext || window.webkitAudioContext; if (!AudioCtx) return false; if (!cosmicAudio.context) { const ctx = new AudioCtx(); cosmicAudio.context = ctx; cosmicAudio.master = ctx.createGain(); cosmicAudio.master.gain.value = cosmicAudio.volume; cosmicAudio.master.connect(ctx.destination); cosmicAudio.musicGain = ctx.createGain(); cosmicAudio.musicGain.gain.value = 0.18; cosmicAudio.musicGain.connect(cosmicAudio.master); cosmicAudio.engineGain = ctx.createGain(); cosmicAudio.engineGain.gain.value = 0; cosmicAudio.engineGain.connect(cosmicAudio.master); } if (cosmicAudio.context.state === "suspended") { cosmicAudio.context.resume(); } return true; } function cosmicTone(freq, duration, type = "sine", volume = 0.15, endFreq = null, destination = null) { if (!cosmicAudio.enabled || !initCosmicAudio()) return; const ctx = cosmicAudio.context; const now = ctx.currentTime; const osc = ctx.createOscillator(); const gain = ctx.createGain(); osc.type = type; osc.frequency.setValueAtTime(Math.max(1, freq), now); if (endFreq !== null) { osc.frequency.exponentialRampToValueAtTime( Math.max(1, endFreq), now + duration ); } gain.gain.setValueAtTime(Math.max(0.001, volume), now); gain.gain.exponentialRampToValueAtTime(0.001, now + duration); osc.connect(gain); gain.connect(destination || cosmicAudio.master); osc.start(now); osc.stop(now + duration + 0.02); } // Nhac nen vu tru huyen bi function startCosmicMusic() { if (!initCosmicAudio() || cosmicAudio.musicTimer) return; const notes = [55, 82.41, 110, 73.42, 98, 65.41]; let index = 0; const playNote = () => { if (!cosmicAudio.enabled || !cosmicAudio.musicTimer) return; const freq = notes[index % notes.length]; cosmicTone(freq, 2.8, "sine", 0.055, freq * 1.01, cosmicAudio.musicGain); cosmicTone(freq * 1.5, 2.2, "sine", 0.018, freq * 1.48, cosmicAudio.musicGain); index++; }; cosmicAudio.musicTimer = true; playNote(); // Lap lai nhac nen khi game dang bat cosmicAudio.musicInterval = setInterval(playNote, 2600); } function stopCosmicMusic() { if (cosmicAudio.musicInterval) { clearInterval(cosmicAudio.musicInterval); cosmicAudio.musicInterval = null; } cosmicAudio.musicTimer = null; } // Dong co nang luong nhe function startCosmicEngine() { if (!initCosmicAudio() || cosmicAudio.engineOsc) return; const ctx = cosmicAudio.context; const osc = ctx.createOscillator(); osc.type = "triangle"; osc.frequency.value = 48; osc.connect(cosmicAudio.engineGain); osc.start(); cosmicAudio.engineOsc = osc; cosmicAudio.engineGain.gain.setTargetAtTime( 0.025, ctx.currentTime, 0.25 ); } function stopCosmicEngine() { if (!cosmicAudio.context) return; cosmicAudio.engineGain.gain.setTargetAtTime( 0, cosmicAudio.context.currentTime, 0.12 ); if (cosmicAudio.engineOsc) { cosmicAudio.engineOsc.stop( cosmicAudio.context.currentTime + 0.5 ); cosmicAudio.engineOsc = null; } } // Va cham thien thach function soundCollision() { cosmicTone(130, 0.35, "triangle", 0.30, 38); cosmicTone(65, 0.28, "sine", 0.22, 30); } // Dat cap do moi function soundLevelUp() { cosmicTone(440, 0.16, "sine", 0.18, 660); setTimeout(() => cosmicTone(660, 0.22, "sine", 0.18, 880), 120); } // Game over function soundGameOver() { stopCosmicMusic(); stopCosmicEngine(); cosmicTone(220, 0.55, "triangle", 0.24, 110); setTimeout(() => cosmicTone(110, 0.9, "sine", 0.20, 45), 300); } // Bat/tat am thanh function setCosmicAudioEnabled(enabled) { cosmicAudio.enabled = enabled; if (!cosmicAudio.context) return; if (!enabled) { stopCosmicMusic(); stopCosmicEngine(); cosmicAudio.master.gain.setTargetAtTime( 0, cosmicAudio.context.currentTime, 0.05 ); } else { cosmicAudio.master.gain.setTargetAtTime( cosmicAudio.volume, cosmicAudio.context.currentTime, 0.05 ); } }
